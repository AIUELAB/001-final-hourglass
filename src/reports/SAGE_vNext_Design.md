### 4.1 Inventory Manager
```python
class InventoryManager:
    TARGET_PER_AGE = 365
    UPPER_LEVEL_CRITERIA = {
        'min_factual_density': 7.0,
        'min_generation_quality': 8.0
    }

    def get_status(self, age: int) -> dict:
        """年齢別の達成状況を返す"""
        return {
            'age': age,
            'current': self.count_upper_level(age),
            'target': self.TARGET_PER_AGE,
            'achieved': self.count_upper_level(age) >= self.TARGET_PER_AGE,
            'mode': 'STOP' if achieved else 'GENERATE'
        }

    def should_generate(self, age: int) -> bool:
        """生成すべきかどうか"""
        return not self.get_status(age)['achieved']

    def should_replace(self, age: int, new_score: float) -> bool:
        """置換すべきかどうか（5%以上改善）"""
        min_score = self.get_min_upper_score(age)
        return new_score > min_score * 1.05
```

### 4.2 Anti-Gaming Monitor
```python
class AntiGamingMonitor:
    def check(self, episode: dict) -> tuple[bool, list[str]]:
        violations = []

        # 1. キーワード密度チェック
        if self._keyword_density(episode['text']) > 0.15:
            violations.append('keyword_stuffing')

        # 2. 固有名詞/年号必須
        if not self._has_proper_nouns(episode['text']):
            violations.append('missing_specifics')

        # 3. 抽象語比率
        if self._abstract_ratio(episode['text']) > 0.3:
            violations.append('too_abstract')

        # 4. N-gram重複
        if self._ngram_repetition(episode['text']) > 0.2:
            violations.append('template_smell')

        return len(violations) == 0, violations
```
